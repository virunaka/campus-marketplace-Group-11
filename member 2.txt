import React, { useState } from "react";
import { Pressable, ScrollView, StyleSheet, Text, TextInput, View } from "react-native";
import { Listing, MockUser } from "../types";

export function AdminPanelPage({
  listings,
  students,
  onAddListing,
  onDeleteListing,
  onRegisterStudent,
  onDeleteStudent,
}: {
  listings: Listing[];
  students: MockUser[];
  onAddListing: (title: string, price: number, category: string) => void;
  onDeleteListing: (id: string) => void;
  onRegisterStudent: (name: string, email: string, campus: string) => void;
  onDeleteStudent: (id: string) => void;
}) {
  const [newTitle, setNewTitle] = useState("");
  const [newPrice, setNewPrice] = useState("");
  const [newCategory, setNewCategory] = useState("Tech");

  const [studentName, setStudentName] = useState("");
  const [studentEmail, setStudentEmail] = useState("");
  const [studentCampus, setStudentCampus] = useState("North Campus");

  return (
    <ScrollView contentContainerStyle={styles.container}>
      <Text style={styles.heading}>Admin Dashboard</Text>
      <Text style={styles.subheading}>Manage listings and registered students</Text>

      {/* Add Listing Section */}
      <View style={styles.card}>
        <Text style={styles.cardTitle}>Add New Listing (Admin)</Text>
        <TextInput placeholder="Item Title" value={newTitle} onChangeText={setNewTitle} style={styles.input} />
        <TextInput placeholder="Price ($)" value={newPrice} onChangeText={setNewPrice} keyboardType="numeric" style={styles.input} />
        <Pressable
          style={styles.button}
          onPress={() => {
            if (newTitle && newPrice) {
              onAddListing(newTitle, Number(newPrice), newCategory);
              setNewTitle("");
              setNewPrice("");
            }
          }}
        >
          <Text style={styles.buttonText}>+ Add Listing</Text>
        </Pressable>
      </View>

      {/* Remove Listings Section */}
      <View style={styles.card}>
        <Text style={styles.cardTitle}>All Listings ({listings.length})</Text>
        {listings.map((item) => (
          <View key={item.id} style={styles.listItem}>
            <View style={{ flex: 1 }}>
              <Text style={styles.itemTitle}>{item.title}</Text>
              <Text style={styles.itemMuted}>${item.price} · Seller: {item.seller}</Text>
            </View>
            <Pressable style={styles.deleteButton} onPress={() => onDeleteListing(item.id)}>
              <Text style={styles.deleteText}>Remove</Text>
            </Pressable>
          </View>
        ))}
      </View>

      {/* Register Student Section */}
      <View style={styles.card}>
        <Text style={styles.cardTitle}>Register New Student</Text>
        <TextInput placeholder="Student Name" value={studentName} onChangeText={setStudentName} style={styles.input} />
        <TextInput placeholder="Student Email" value={studentEmail} onChangeText={setStudentEmail} style={styles.input} />
        <Pressable
          style={styles.button}
          onPress={() => {
            if (studentName && studentEmail) {
              onRegisterStudent(studentName, studentEmail, studentCampus);
              setStudentName("");
              setStudentEmail("");
            }
          }}
        >
          <Text style={styles.buttonText}>+ Register Student</Text>
        </Pressable>
      </View>

      {/* Remove Students Section */}
      <View style={styles.card}>
        <Text style={styles.cardTitle}>Registered Students ({students.length})</Text>
        {students.map((st) => (
          <View key={st.id} style={styles.listItem}>
            <View style={{ flex: 1 }}>
              <Text style={styles.itemTitle}>{st.name}</Text>
              <Text style={styles.itemMuted}>{st.email} · {st.campus}</Text>
            </View>
            <Pressable style={styles.deleteButton} onPress={() => onDeleteStudent(st.id)}>
              <Text style={styles.deleteText}>Remove</Text>
            </Pressable>
          </View>
        ))}
      </View>
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { padding: 20, paddingBottom: 110 },
  heading: { color: "#173C34", fontSize: 26, fontWeight: "800" },
  subheading: { color: "#87918C", fontSize: 13, marginBottom: 20 },
  card: { backgroundColor: "#FFF", borderRadius: 16, padding: 18, marginBottom: 20, borderWidth: 1, borderColor: "#E9ECE6" },
  cardTitle: { color: "#173C34", fontSize: 16, fontWeight: "700", marginBottom: 12 },
  input: { height: 44, borderWidth: 1, borderColor: "#DDE4DC", borderRadius: 10, paddingHorizontal: 12, marginBottom: 10 },
  button: { height: 44, backgroundColor: "#1F5D4C", borderRadius: 10, justifyContent: "center", alignItems: "center" },
  buttonText: { color: "#FFF", fontWeight: "700", fontSize: 13 },
  listItem: { flexDirection: "row", justifyContent: "space-between", alignItems: "center", paddingVertical: 10, borderBottomWidth: 1, borderBottomColor: "#F0F3EF" },
  itemTitle: { color: "#173C34", fontWeight: "600", fontSize: 13 },
  itemMuted: { color: "#87918C", fontSize: 11, marginTop: 2 },
  deleteButton: { backgroundColor: "#FBECEE", paddingHorizontal: 10, paddingVertical: 6, borderRadius: 6 },
  deleteText: { color: "#B64950", fontSize: 11, fontWeight: "700" },
});
